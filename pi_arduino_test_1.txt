import serial
import time

arduino = serial.Serial('/dev/ttyUSB0', 115200, timeout=2)

time.sleep(2)

# Read startup message
startup = arduino.readline().decode().strip()
print("Arduino:", startup)

# Send test message
arduino.write(b"How are you?\n")

# Read response
response = arduino.readline().decode().strip()

print("Arduino replied:", response)
